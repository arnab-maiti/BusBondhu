/**
 * Format duration minutes into "Xh Ym"
 */
export function formatDuration(minutes) {
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  return m > 0 ? `${h}h ${m}m` : `${h}h`;
}

/**
 * Search buses matching criteria.
 * Returns a Promise to simulate asynchronous network retrieval.
 */
export async function searchBuses({ from, to, date, passengers = 1, simulateError = false }) {
  // Simulate network latency (350ms - 550ms)
  await new Promise((resolve) => setTimeout(resolve, 450));

  if (simulateError) {
    throw new Error("Unable to load bus schedules. Please verify your connection and try again.");
  }

  const origin = (from || "Kolkata").trim();
  const destination = (to || "Durgapur").trim();

  // If origin equals destination, return empty results intentionally
  if (origin.toLowerCase() === destination.toLowerCase()) {
    return {
      results: [],
      origin,
      destination,
      date: date || new Date().toISOString().split("T")[0],
      passengers: Number(passengers) || 1,
      isDemoFixture: true,
      notice: DEMO_NOTICE,
    };
  }

  // Construct realistic bus schedules based on the search query
  const results = BUS_TEMPLATES.map((tmpl, idx) => {
    const operator = OPERATORS[tmpl.operatorIndex % OPERATORS.length];
    const durationStr = formatDuration(tmpl.durationMinutes);

    return {
      id: `${tmpl.id}-${origin.slice(0, 3).toUpperCase()}-${idx + 1}`,
      busNumber: `WB-${20 + (idx % 15)}-${1000 + idx * 87}`,
      operatorName: operator.name,
      operatorType: operator.type,
      rating: operator.rating,
      reviewsCount: operator.reviews,
      busType: tmpl.busType,
      category: tmpl.category,
      isAc: tmpl.isAc,
      departureTime: tmpl.depTime,
      arrivalTime: tmpl.arrTime,
      duration: durationStr,
      durationMinutes: tmpl.durationMinutes,
      originCity: origin,
      destinationCity: destination,
      boardingPoint: `${origin} (${tmpl.boardingPoint})`,
      droppingPoint: `${destination} (${tmpl.droppingPoint})`,
      fare: tmpl.baseFare,
      seatsTotal: tmpl.seatsTotal,
      seatsAvailable: tmpl.seatsAvailable,
      amenities: tmpl.amenities,
      isLiveTrackable: tmpl.amenities.includes("live_tracking"),
      isDemoFixture: true,
    };
  });

  return {
    results,
    origin,
    destination,
    date: date || new Date().toISOString().split("T")[0],
    passengers: Number(passengers) || 1,
    isDemoFixture: true,
    notice: DEMO_NOTICE,
  };
}

/**
 * BusBondhu Seat Availability Service
 * 
 * Provides seat layout and availability fixtures for the Seat Selection UI.
 * In Phase 3, this returns clearly labeled development fixtures for UI demonstration.
 * Never presents fixture availability as live inventory.
 */

export const SEAT_FIXTURE_NOTICE = 
  "Development fixture: Seat availability and layout shown below are for UI demonstration only and do not reflect live booking inventory.";

export const MAX_SEATS_PER_BOOKING = 6;

/**
 * Generate a deterministic bus seat map based on tripId
 */
export function getBusSeatLayout(tripId) {
  // Deterministic seed from tripId
  let hash = 0;
  for (let i = 0; i < (tripId || "").length; i++) {
    hash = (hash << 5) - hash + tripId.charCodeAt(i);
    hash |= 0;
  }
  const absHash = Math.abs(hash);

  const rows = 10;
  const seats = [];

  for (let r = 1; r <= rows; r++) {
    // Left side: A (Window), B (Aisle)
    // Right side: C (Aisle), D (Window)
    const columns = [
      { col: "A", type: "Window", side: "left" },
      { col: "B", type: "Aisle", side: "left" },
      { col: "C", type: "Aisle", side: "right" },
      { col: "D", type: "Window", side: "right" },
    ];

    columns.forEach(({ col, type, side }) => {
      const seatNumber = `${r}${col}`;
      // Deterministically mark ~30% of seats as booked/unavailable
      const seedVal = (absHash + r * 17 + col.charCodeAt(0) * 11) % 100;
      const isUnavailable = seedVal < 32;

      seats.push({
        id: seatNumber,
        row: r,
        col,
        type,
        side,
        isUnavailable,
      });
    });
  }

  return {
    tripId: tripId || "TRIP-DEFAULT",
    rows,
    seats,
    totalSeats: seats.length,
    availableSeatsCount: seats.filter((s) => !s.isUnavailable).length,
    isDemoFixture: true,
    notice: SEAT_FIXTURE_NOTICE,
  };
}
